\begin{frame}[t,label=md-42]{Example Scientific Loop}
\label{md-62}
\small
\slideheading{Illustrative workflow; not an established discovery}
\smallskip
\begin{enumerate}
\setlength{\itemsep}{0.45em}
    \item \textbf{Literature:} Find a reaction, key energy regions, and conflicting evidence.
    \item \textbf{Hypothesis:} Propose competing resonance mechanisms.
    \item \textbf{Code:} Reproduce calculations and predict cross sections and astrophysical consequences.
    \item \textbf{Experiment:} Measure at JUNA; extract cross sections, rates, and uncertainty.
    \item \textbf{Follow-up:} Code proposes a discriminating test; run targeted
    simulations or analyze new / archival observations after coordinated execution.
    \item \textbf{Interpretation:} Check results independently; return new constraints and unresolved questions to the next cycle.
\end{enumerate}
\end{frame}
