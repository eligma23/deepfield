\subsection{Research Loop and Multi-Agent Architecture}
% Source page: Deepfield.md Page 42; exactly one frame.
\setsource{\href{https://doi.org/10.1038/s41586-026-10652-y}{Robin, Nature (2026)}}
\begin{frame}[t,label=md-42]{Research Loop}
\small
\slideheading{One Loop connects two experiment types}
\smallskip
\begin{enumerate}
\setlength{\itemsep}{0.45em}
    \item \textbf{Literature:} Identify reactions, energy regions, and evidence
    under scientist-defined goals and constraints.
    \item \textbf{Hypothesis + Code:} Develop competing mechanisms and
    quantitative predictions using verified, reusable tools.
    \item \textbf{First experiment:} Experiment Agent and the human team
    optimize and perform a real cross-section measurement.
    \item \textbf{Code-led follow-up:} Infer cross sections and rates;
    propose discriminating observations, archival analyses, or simulations.
    \item \textbf{Execution:} Code runs simulations and analyzes data;
    Experiment coordinates new observations with scientists and observing teams.
    \item \textbf{Interpretation and next cycle:} Independently verify results;
    scientists judge explanations and update evidence, models, and questions.
\end{enumerate}
\end{frame}

\begin{frame}[c,label=md-43]{Core MAS Roles}
\footnotesize
\renewcommand{\arraystretch}{1.15}
\begin{tabular*}{\textwidth}{@{\extracolsep{\fill}}L{0.18\textwidth}L{0.75\textwidth}@{}}
\toprule
\textbf{Role} & \textbf{Main responsibility} \\
\midrule
\textbf{Literature} & Evidence, conflicts, knowledge gaps, reactions and energy regions;
reusable data, code, and methods. \\[0.5ex]
\textbf{Hypothesis} & Generate, debate, refine, and select competing mechanisms;
predictions, falsification, and feedback from both experiment types. \\[0.5ex]
\textbf{Code} & Reuse, adapt, or develop tools; infer cross sections and rates;
run astrophysical simulations and proactively propose follow-ups. \\[0.5ex]
\textbf{Experiment} & Feasible designs, beam measurements, and instrument feedback;
coordinate new observations with scientists and observing teams. \\[0.5ex]
\textbf{Verification} & Independent evidence, physical and numerical checks;
uncertainty, reproducibility, and external validation. \\[0.5ex]
\textbf{Supervisor} & Tasks, shared state, memory, dependencies, resources,
and recovery. \\[0.5ex]
\bottomrule
\end{tabular*}\par

\slidenote{Scientists set and revise goals and judge scientific conclusions.
Research directions define capabilities; Agent roles define responsibilities.}
\end{frame}
