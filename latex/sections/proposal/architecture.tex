\subsection{Research Loop and Multi-Agent Architecture}
% Source page: Deepfield.md Page 42; exactly one frame.
\setsource{\href{https://doi.org/10.1038/s41586-026-10652-y}{Robin, Nature (2026)}}
\begin{frame}[t,label=md-42]{Research Loop}
\small
\slideheading{One Loop connects two experiment types}
\smallskip
\begin{enumerate}
\setlength{\itemsep}{0.45em}
    \item \textbf{Literature:} Set goals and constraints; identify reactions, energy regions, and evidence.
    \item \textbf{Hypothesis + Code:} Propose mechanisms and reuse code for predictions.
    \item \textbf{First experiment:} Optimize the plan and measure physical cross sections.
    \item \textbf{Code-led follow-up:} Analyze data; proactively propose and execute observations or simulations.
    \item \textbf{Interpretation:} Joint analysis, independent verification, and scientist judgment.
    \item \textbf{Next cycle:} Update evidence, models, and hypotheses.
\end{enumerate}
\end{frame}
