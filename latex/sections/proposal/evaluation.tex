\begin{frame}[t,label=md-61]{Independent Evaluation and Ablation}
\small
\begin{itemize}
\setlength{\itemsep}{0.65em}
    \item \textbf{Independent tests:} Hold out by publication time, reaction system, or experimental batch; check overlap.
    \item \textbf{Baselines:} Human workflows, baseline Agents, full MAS; match resource budgets.
    \item \textbf{Ablations:} Tools, memory, fusion, and candidate roles; include coordination cost.
    \item \textbf{Key comparison:} Measurement and analysis alone versus the full Loop with proactive follow-ups.
    \item \textbf{Report:} Physics checks, calibrated uncertainty, evidence, human effort, and costs.
\end{itemize}

\slidenote{Separate replay, simulation, and prospective physical results.
Internal rankings and analysis consensus are not independent evidence.}
\end{frame}

\begin{frame}[t,label=md-62]{Example Scientific Loop}
\small
\slideheading{Illustrative workflow; not an established discovery}
\smallskip
\begin{enumerate}
\setlength{\itemsep}{0.45em}
    \item \textbf{Literature:} Find a reaction, key energy regions, and conflicting evidence.
    \item \textbf{Hypothesis:} Propose competing resonance mechanisms.
    \item \textbf{Code:} Reproduce calculations and predict cross sections and astrophysical consequences.
    \item \textbf{Experiment:} Measure at JUNA; extract cross sections, rates, and uncertainty.
    \item \textbf{Follow-up:} Code proposes a discriminating test; run targeted
    simulations or analyze new / archival observations after coordinated execution.
    \item \textbf{Interpretation:} Check results independently; return new constraints and unresolved questions to the next cycle.
\end{enumerate}
\end{frame}
