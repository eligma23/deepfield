\begin{frame}[c,label=md-60]{VI. Scientific Discovery Benchmarks}
\small
\renewcommand{\arraystretch}{1.25}
\begin{tabular*}{\textwidth}{@{\extracolsep{\fill}}L{0.23\textwidth}L{0.70\textwidth}@{}}
\toprule
\textbf{Task} & \textbf{Observable evaluation criteria} \\
\midrule
\textbf{Literature} & Evidence coverage and citation support; numerical,
unit, and figure-reading accuracy; conflict detection. \\[1.1ex]
\textbf{Hypotheses} & Independently checked novelty and physical plausibility;
quantitative, discriminating predictions; revision after new evidence. \\[1.1ex]
\textbf{Analysis and simulation} & Reference error, convergence, physical
consistency, uncertainty coverage, and reproducibility. \\[1.1ex]
\textbf{Experiments} & Feasible plans, measurement precision per beam hour,
reliable execution, and realized scientific benefit. \\
\bottomrule
\end{tabular*}\par
\medskip
\textbf{Full-loop outcome:} Additional verified findings, stronger constraints,
or better discrimination between physical explanations.
\slidenote{Set task-specific tolerances with experts before evaluation;
use raw data, reference calculations, and external evidence.}
\note{Benchmark the complete outputs of literature, hypothesis, code, and
experimental workflows. Report continuous errors and completion rates, not
only a single aggregate score. Check uncertainty coverage on cases with known
truth; real unknown quantities require independent empirical constraints.
Compare realized information gain or prediction changes with design forecasts.
Negative or null results count when they provide verified scientific constraints.
Claims of novelty require independent literature checks; internal rankings and
LLM confidence are insufficient.}
\end{frame}

\begin{frame}[t,label=md-61]{Independent Evaluation and Ablation}
\small
\begin{itemize}
\setlength{\itemsep}{0.65em}
    \item \textbf{Independent tests:} Hold out by publication time, reaction system, or experimental batch; check overlap.
    \item \textbf{Baselines:} Human workflows, baseline Agents, full MAS; match resource budgets.
    \item \textbf{Ablations:} Tools, memory, fusion, and candidate roles; include coordination cost.
    \item \textbf{Key comparison:} Measurement and analysis alone versus the full Loop with proactive follow-ups.
    \item \textbf{Report:} Physics checks, calibrated uncertainty, evidence, human effort, and costs.
\end{itemize}

\slidenote{Separate replay, simulation, and prospective physical results.
Internal rankings and analysis consensus are not independent evidence.}
\end{frame}
