\begin{frame}[c,label=md-39]{Deepfield Research Proposal}
\small
\begin{block}{An AI-Driven, Human-in-the-Loop Laboratory}
\textbf{Setting:} JUNA / Jinping deep underground nuclear astrophysics experiments.
\end{block}
\medskip
\textbf{Research Vision}
\begin{enumerate}
    \renewcommand{\theenumi}{\arabic{enumi}}
    \setbeamertemplate{enumerate item}{\insertenumlabel}
    \setlength{\itemsep}{0.5em}
    \item \textbf{Agentic research:} Coordinate literature, hypotheses, code,
    experiments, and analysis.
    \item \textbf{Human guidance:} Scientists set goals, contribute expertise,
    and judge key decisions.
    \item \textbf{Recursive improvement:} Study feedback-driven updates to
    knowledge, code, strategies, and model capabilities.
    \item \textbf{Scientific feedback:} Use measurements, follow-ups, and
    independent checks to refine interpretations.
    \item \textbf{Long-term aim:} Accumulate research experience through
    human--AI collaboration.
\end{enumerate}
\end{frame}

\small
\slideheading{For one scientist-confirmed reaction question}
\smallskip
\begin{enumerate}
\setlength{\itemsep}{0.55em}
    \item Identify evidence-grounded energy regions and competing hypotheses.
    \item Complete a physical cross-section measurement and reproducible analysis.
    \item Have the Code Agent proactively propose a follow-up observation or simulation.
    \item Execute the follow-up and independently check its results.
    \item Compare astrophysical conclusions before and after measurement.
\end{enumerate}

\slidenote{Replay supports development; prospective acceptance needs a
physical measurement. Unexecuted observing applications remain pending.
