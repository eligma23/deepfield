\note{The name connects nuclear physics and astrophysics, with depth and
ground also evoking Jinping's deep underground laboratory. The Hubble Deep
Field provides a second inspiration: in 1995, Robert Williams proposed
observing an apparently empty patch of sky in Ursa Major. Ten days of
observations revealed about 3,000 galaxies. The connection to Deepfield is
the move from studying known targets toward exploring unknown regions,
asking how search strategies and resource allocation shape discovery.}

\begin{frame}[c,label=md-39]{Deepfield Research Proposal}
\small
\begin{block}{An AI-Driven, Human-in-the-Loop Laboratory}
\textbf{Setting:} JUNA / Jinping deep underground nuclear astrophysics experiments.
\end{block}
\smallskip
\textbf{Research Vision}
\begin{enumerate}
    \renewcommand{\theenumi}{\arabic{enumi}}
    \setbeamertemplate{enumerate item}{\insertenumlabel}
    \setlength{\itemsep}{0.3em}
    \item \textbf{Autonomous research:} Investigate literature, generate
    hypotheses, reuse and develop code, and design and analyze experiments.
    \item \textbf{Human-in-the-loop:} Scientists conduct experiments;
    the full research process remains understandable, inspectable, and explainable.
    \item \textbf{Scientific loop:} Connect literature, hypotheses, simulations,
    nuclear experiments, and astronomical observations in one iterative loop.
    \item \textbf{Recursive improvement:} Learn better research
    strategies and recursively improve agent capability and efficiency by meta-learning and iteration.
    \item \textbf{Long-term aim:} Progress from semi-autonomous to autonomous
    research, and explore the evolution and emergence of scientific intelligence.
\end{enumerate}
\end{frame}

\begin{frame}[t,label=md-41]{Acceptance Criteria}
\small
\slideheading{For one scientist-confirmed reaction question}
\smallskip
\begin{enumerate}
\setlength{\itemsep}{0.55em}
    \item Improve on our current findings with additional promising energy
    regions or more informative competing hypotheses.
    \item Complete a physical cross-section measurement and reproducible analysis.
    \item Have the system proactively propose discriminating experiments,
    astronomical observations, archival analyses, or simulations.
    \item Complete the follow-up with the responsible teams and
    independently verify its results.
    \item Compare astrophysical conclusions before and after measurement.
\end{enumerate}

\slidenote{\textbf{Acceptance requires:} Physical consistency checks
(orders of magnitude, dimensions, conservation laws, etc.); real experimental
validation; and additional, independently verified scientific findings.}
\end{frame}
