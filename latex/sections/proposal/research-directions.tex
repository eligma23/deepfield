\begin{frame}[t,label=md-55]{I. Continual Knowledge-Guided Exploration}
\small
\begin{itemize}
\setlength{\itemsep}{0.45em}
    \item \textbf{Continual evidence alignment:} Integrate evolving literature,
    nuclear data, and experiments; retain provenance, validity, conflicts,
    and candidate reactions or energy regions.
    \item \textbf{Scientific world models:} Discover and revise models that
    make quantitative predictions, using physically constrained structure,
    equation, or program search; Direction III implements and tests them.
    \item \textbf{Experimental experience:} Link configurations, raw data,
    analyses, anomalies, and successful or failed procedures to hypotheses
    and physical models.
    \item \textbf{Long-term research state:} Maintain tasks, dependencies,
    hypothesis branches, pending experiments, and recovery across weeks or months.
\end{itemize}
\slidenote{Evidence records, predictive models, experimental experience,
and task state serve different roles. Test whether continual evidence
alignment improves quantitative research decisions.}
\end{frame}

\begin{frame}[t,label=md-56]{II. Hierarchical Scientific Hypothesis Generation}
\footnotesize
\begin{itemize}
\setlength{\itemsep}{0.15em}
    \item \textbf{Hierarchical doubt:} Question assumptions, approximations,
    validity, and anomalies across measurements, nuclei, networks,
    astrophysics, and fundamental physics.
    \item \textbf{Cross-domain reasoning:} Transfer theories, mathematical
    structures, and mechanisms through analogy and abduction.
    \item \textbf{Simulation-guided discovery:} Use Code scans, sensitivity,
    and uncertainty to reveal dominant factors, model mismatch, and new questions.
    \item \textbf{Mechanisms:} Derive competing causal explanations,
    observable predictions, and their uncertainty.
    \item \textbf{Open-world generation:} Explore unlisted variables,
    mechanisms, and questions; request new or revised models in Direction I.
    \item \textbf{Robustness and falsifiability:} Refine using both experiment
    types; test parameter, model, and uncertainty assumptions and state
    falsification conditions.
    \item \textbf{Quantitative selection:} Combine scientific value, novelty,
    information gain, sensitivity, uncertainty, and feasibility;
    retain complementary hypotheses.
\end{itemize}
\slidenote[0.15cm]{Output: Versioned hypothesis cards with evidence, predictions,
falsification tests, and revisions.}
\end{frame}

\begin{frame}[t,label=md-57]{III. Data Analysis \& Simulation with Codes}
\small
\slideheading{Reuse first}
\smallskip

Search $\rightarrow$ reproduce $\rightarrow$ check validity $\rightarrow$ reuse
$\rightarrow$ adapt or extend $\rightarrow$ fill remaining gaps $\rightarrow$ validate.
\medskip
\begin{itemize}
\setlength{\itemsep}{0.5em}
    \item Reproduce and reuse reaction, network, stellar, and analysis
    software; verify reference results and validity.
    \item Adapt or extend first; develop new code only for unmet needs,
    guided by execution feedback and numerical / physical checks.
    \item Infer cross sections and rates; check calibration, background,
    efficiency, and statistical / systematic uncertainty.
    \item After analysis, \textbf{proactively propose discriminating
    observations, archival analyses, or targeted simulations}.
    \item Run astrophysical simulations and analyze observations; archive
    code, environments, provenance, figures, and research reports.
\end{itemize}
\end{frame}

\begin{frame}[t,label=md-58]{IV. Experimental Design \& Optimization}
\small
\begin{itemize}
\setlength{\itemsep}{0.65em}
    \item \textbf{Formulate:} Model observables, competing hypotheses,
    uncertainty, and constraints; target information gain or parameter precision.
    \item \textbf{Design offline:} Use Bayesian design and multi-objective
    optimization for energies, beam, target, detectors, time, and resources.
    \item \textbf{Simulate:} Background, efficiency, response, feasibility, and objective trade-offs.
    \item \textbf{Adapt online:} Bayesian updates and optimization using measurements and scientist feedback.
    \item \textbf{Prepare execution:} Validate interfaces in simulation or
    replay; assess Code-proposed follow-ups and coordinate observations
    with scientists and observing teams.
\end{itemize}

\medskip
\textbf{Output:} Executable plans, configurations, calibration, and adjustment strategies; Direction V executes them.
\end{frame}

\begin{frame}[t,label=md-59]{V. Experimentation Implementation}
\small
\begin{itemize}
\setlength{\itemsep}{0.9em}
    \item \textbf{Beam experiments:} Experiment Agent and the human team
    perform real measurements, calibration, background studies, and repetitions.
    \item \textbf{Astrophysical experiments:} Carry out Code-proposed
    observations with scientists and observing teams, or analyze independent
    archives to test nuclear-constrained predictions.
    \item \textbf{Numerical experiments:} Code Agent reuses or extends
    reaction and stellar codes for controlled variations, competing and
    counterfactual models, and uncertainty propagation.
\end{itemize}

\medskip
Beam measurement is the first experiment type; the post-measurement
observation or simulation is the second.
\slidenote{Preserve records and negative results; return supported,
excluded, and unresolved explanations to the next cycle. Distinguish
simulation deductions from independent observational evidence.}
\end{frame}
