\begin{frame}[t,label=md-55]{I. Continual Knowledge-Guided Exploration}
\small
\begin{itemize}
\setlength{\itemsep}{0.65em}
    \item \textbf{Align changing evidence:} Link literature, nuclear data,
    measurements, and observations; reconcile units, conditions, and conflicting results.
    \item \textbf{Discover physical models:} Search physically constrained
    equations or programs; revise mechanisms when existing models fail quantitative tests.
    \item \textbf{Learn from experience:} Connect configurations, anomalies,
    successful procedures, and failures to their conditions and scientific outcomes.
    \item \textbf{Sustain research:} Preserve hypothesis branches, task
    dependencies, and pending experiments; resume work as new evidence arrives.
    \item \textbf{Test decision value:} Compare fixed and evolving knowledge
    on reaction priorities, model predictions, and proposed follow-up measurements.
\end{itemize}
\slidenote{Output: traceable evidence, executable predictive models,
and reusable research experience.}
\note{Evidence records describe what has been reported; a scientific world
model maps physical assumptions and inputs to quantitative predictions.
Use nuclear reaction and astrophysical models as a starting point, extending
structure only where data and physical checks justify it. Compare evidence
updates, experience reuse, and model revisions separately on held-out reactions.
The aim is improved scientific decisions, rather than database growth alone.}
\end{frame}

\begin{frame}[t,label=md-56]{II. Hierarchical Scientific Hypothesis Generation}
\footnotesize
\begin{itemize}
\setlength{\itemsep}{0.15em}
    \item \textbf{Hierarchical doubt:} Question assumptions, approximations,
    validity, and anomalies across measurements, nuclei, networks,
    astrophysics, and fundamental physics.
    \item \textbf{Cross-domain reasoning:} Transfer theories, mathematical
    structures, and mechanisms through analogy and abduction.
    \item \textbf{Simulation-guided discovery:} Use Code scans, sensitivity,
    and uncertainty to reveal dominant factors, model mismatch, and new questions.
    \item \textbf{Mechanisms:} Derive competing causal explanations,
    observable predictions, and their uncertainty.
    \item \textbf{Open-world generation:} Explore unlisted variables,
    mechanisms, and questions; request new or revised models in Direction I.
    \item \textbf{Robustness and falsifiability:} Refine using both experiment
    types; test parameter, model, and uncertainty assumptions and state
    falsification conditions.
    \item \textbf{Quantitative selection:} Combine scientific value, novelty,
    information gain, sensitivity, uncertainty, and feasibility;
    retain complementary hypotheses.
\end{itemize}
\slidenote[0.15cm]{Output: Versioned hypothesis cards with evidence, predictions,
falsification tests, and revisions.}
\end{frame}

\begin{frame}[t,label=md-57]{III. Data Analysis and Scientific Simulation}
\small
\slideheading{From reproducible computation to new questions}
\smallskip
Discover $\rightarrow$ reproduce $\rightarrow$ reuse / extend $\rightarrow$ validate.
\medskip
\begin{itemize}
\setlength{\itemsep}{0.55em}
    \item \textbf{Build on existing software:} Reproduce reference cases;
    adapt validated tools and develop new code where capabilities are missing.
    \item \textbf{Infer nuclear quantities:} Model calibration, background,
    and efficiency; extract cross sections and rates with correlated uncertainties.
    \item \textbf{Connect to astrophysics:} Propagate rates through reaction
    networks and stellar models; quantify changes in abundances and observables.
    \item \textbf{Propose follow-ups:} Identify unresolved mechanisms and
    suggest discriminating observations, archival analyses, or controlled simulations.
    \item \textbf{Deliver reproducible results:} Preserve data, code,
    environments, physical checks, and the calculations behind each conclusion.
\end{itemize}
\note{Use execution feedback and independent reference cases to improve code.
Check dimensions, conservation laws, numerical convergence, validity ranges,
and known limits. Sensitivity ranks influential inputs; combine it with their
uncertainties to prioritize new measurements. Account for astrophysical nuisance
parameters when interpreting nuclear-rate changes. Validate surrogate models
on independent simulations before using them for inference or experimental
design. Follow-up proposals arise from unresolved scientific questions rather
than a mandatory extra step after every analysis.}
\end{frame}

\begin{frame}[t,label=md-58]{IV. Experimental Design and Optimization}
\small
\begin{itemize}
\setlength{\itemsep}{0.65em}
    \item \textbf{Define the scientific utility:} Target model discrimination,
    parameter precision, or reduced uncertainty in astrophysical predictions.
    \item \textbf{Choose informative measurements:} Use Bayesian experimental
    design to select energies, observables, and exposure time under realistic constraints.
    \item \textbf{Optimize instrument performance:} Use simulation and Bayesian
    optimization for beam settings, efficiency, background, and resource trade-offs.
    \item \textbf{Adapt from feedback:} Update physical inference with new data;
    revise the next measurement and stop when expected benefit no longer justifies cost.
    \item \textbf{Prepare feasible follow-ups:} Test plans in simulation or
    replay; coordinate calibration, beam time, and observation or archival access.
\end{itemize}
\slidenote{Output: executable plans with predicted scientific benefit,
uncertainty, resource needs, and criteria for adjustment.}
\note{Bayesian experimental design chooses measurements by expected utility,
such as expected information gain about a physical parameter or model.
Bayesian optimization searches expensive black-box objectives, such as
instrument transmission; it can support design but is not synonymous with
information-driven design. SECAR demonstrates instrument optimization and
ML-assisted measurements, not an established autonomous JUNA research loop.
Include statistical and systematic uncertainty, model discrepancy, and
robustness to prior choices. Scientists conduct actual experiments; design
recommendations must remain interpretable and inspectable.}
\end{frame}

\begin{frame}[t,label=md-59]{V. Scientific Experimentation}
\small
\begin{itemize}
\setlength{\itemsep}{0.85em}
    \item \textbf{Nuclear measurements:} Scientists conduct beam experiments
    with agent assistance; acquire calibrated data, background controls,
    and independent checks of cross sections.
    \item \textbf{Astronomical observations:} Test nuclear-constrained
    predictions against new observations or independent archives;
    account for astrophysical and observational uncertainties.
    \item \textbf{Numerical experiments:} Vary rates and mechanisms under
    controlled assumptions; test convergence and propagate uncertainty
    to astrophysical predictions.
    \item \textbf{New experimental approaches (challenge):} Explore novel
    measurement methods and configurations to access previously inaccessible
    reactions or energy regions; test feasibility and scientific benefit.
\end{itemize}
\slidenote{Output: real measurement results, targeted follow-up results,
and independently checked physical interpretations.}
\note{The initial beam measurement and the post-measurement follow-up belong
to one research loop. A follow-up may be a new observation, an independent
archival analysis, or a targeted numerical experiment. Record their evidence
types separately: a simulation tests consequences under assumptions, whereas
observations constrain nature. A proposed observing program is not a completed
observation. Preserve null results, unresolved disagreements, and all raw data
and run conditions needed to reproduce the analysis.}
\end{frame}
