\begin{frame}[c,label=md-63]{Seven-Month Research Roadmap}
\footnotesize
\renewcommand{\arraystretch}{1.15}
\begin{tabular*}{\textwidth}{@{\extracolsep{\fill}}L{0.08\textwidth}L{0.85\textwidth}@{}}
\toprule
\textbf{Month} & \textbf{Focus and milestone} \\
\midrule
1 & \textbf{Literature:} Evidence, confirmed pilot, and beam-time coordination. \\[0.4ex]
2 & \textbf{Code:} Reproduce at least one reference within expert tolerances;
reusable tools and a discriminating follow-up proposal. \\[0.4ex]
3 & \textbf{Hypothesis:} At least two distinguishable explanations;
quantitative predictions and falsification criteria. \\[0.4ex]
4 & \textbf{Experiment:} Approved plan; optimization and instrument-feedback demo. \\[0.4ex]
5 & \textbf{Beam:} Real cross sections and rates; statistical and
correlated systematic uncertainties. \\[0.4ex]
6 & \textbf{Follow-up:} Execute the second experiment; compare
astrophysical conclusions before and after measurement. \\[0.4ex]
7 & \textbf{Integration:} Frozen independent benchmarks, matched-budget
baselines and ablations, research report. \\[0.4ex]
\bottomrule
\end{tabular*}\par

\slidenote[0.15cm]{Develop Code tools before Hypothesis; the scientific Loop
stays iterative. Shared engineering throughout; monthly benchmarks and
frozen versions. Beam scheduling from Month 1. Replay tests capability;
delayed real measurements leave the Loop pending.}
\end{frame}
