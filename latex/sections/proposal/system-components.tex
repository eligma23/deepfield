\begin{frame}[t,label=md-47]{Data Engineering}
\small
\begin{itemize}
\setlength{\itemsep}{0.65em}
    \item \textbf{Acquire:} Papers, formulas, tables, curves, nuclear data, experiments, simulations, and observations.
    \item \textbf{Normalize:} Identifiers, units, energy frames, metadata, and validity ranges.

\begin{frame}[t,label=md-50]{Context Engineering}
\small
\begin{itemize}

\begin{frame}[t,label=md-52]{Multimodal Fusion and Model Training}
\small
\begin{itemize}
\setlength{\itemsep}{0.45em}
    \item \textbf{Align evidence:} Match literature, nuclear data, spectra,
    energy levels and reaction networks, simulations, and observations by reaction,
    energy, and environment.
    \item \textbf{Reason and quantify:} Combine mechanistic reasoning with
    numerical predictions and uncertainty to evaluate competing hypotheses.
    \item \textbf{Establish baselines:} First test existing models, retrieval,
    and scientific tools; compare separate and combined use.
    \item \textbf{Explore training:} With expert-reviewed samples, study domain
    continual pretraining, SFT, and role-specific LoRA / reinforcement learning.
    \item \textbf{Evaluate contributions:} Separate knowledge updates,
    prompt/code strategies, and parameter training; distinguish LLM confidence
    from physical uncertainty.
\end{itemize}
\end{frame}

\begin{frame}[t,label=md-53]{Validation}
\small
\begin{itemize}
\setlength{\itemsep}{0.65em}
    \item \textbf{Software:} Reference reproduction, convergence, and executable regression cases.
    \item \textbf{Physics:} Units, dimensions, scales, conservation, positivity, and model validity.
    \item \textbf{Uncertainty:} Calibration, interval coverage, and stability across reasonable analyses.
    \item \textbf{Evidence:} Independent checks against competing predictions; distinguish simulations from observations.
    \item \textbf{System:} Re-evaluate changes; track outcomes, failures, human input, and resource costs.
\end{itemize}
